[-ビュー定義
  {
    view_name,view_definition
      quiz_full," SELECT q.id,
          q.easy,
          q.normal,
          q.hard,
          q.expert,
          l.lyric AS lyrics,
          string_agg(m.name, ','::text ORDER BY m.sort_order) AS correct_members,
          l.section_name,
          l.seq,
          l.id AS lyrics_id,
          l.sounds_id,
          s.group_name,
          s.song_name
        FROM ((((quizzes q
          JOIN lyrics l ON ((q.lyrics_id = l.id)))
          JOIN sounds s ON ((l.sounds_id = s.id)))
          LEFT JOIN lyric_members lm ON ((l.id = lm.lyric_id)))
          LEFT JOIN members m ON ((lm.member_id = m.id)))
        WHERE ((q.is_active = true) AND (l.is_active = true) AND (s.is_active = true))
        GROUP BY q.id, q.easy, q.normal, q.hard, q.expert, l.lyric, l.section_name, l.seq, l.id, l.sounds_id, s.group_name, s.song_name;"
  }
